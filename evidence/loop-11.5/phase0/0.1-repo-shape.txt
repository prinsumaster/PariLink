# Copy root workspace config
COPY package.json package-lock.json ./
# Copy workspace package.json files needed for resolution
COPY apps/web/package.json ./apps/web/package.json
COPY packages/plugin-sdk/package.json ./packages/plugin-sdk/package.json
# Install all workspace dependencies
RUN npm ci --workspace=apps/web --workspace=packages/plugin-sdk --include-workspace-root || npm install --legacy-peer-deps

# Builder Stage
FROM base AS builder
COPY --from=deps /app/node_modules ./node_modules
# Copy plugin-sdk source
COPY packages/plugin-sdk ./packages/plugin-sdk
# Build plugin-sdk first
RUN cd packages/plugin-sdk && npx tsc || true
# Copy web app source
COPY apps/web ./apps/web
# Build Next.js
ENV NEXT_TELEMETRY_DISABLED=1
RUN cd apps/web && npm run build

# Runner Stage
FROM base AS runner
ENV NODE_ENV=production
ENV NEXT_TELEMETRY_DISABLED=1

    adduser --system --uid 1001 nextjs

USER nestjs

# Health Check
HEALTHCHECK --interval=30s --timeout=10s --start-period=30s --retries=3 \
  CMD wget --no-verbose --tries=1 --spider http://localhost:${PORT:-3000}/api/v1/health/readiness || exit 1

EXPOSE 3000

COPY --chown=nestjs:nodejs start.sh ./
RUN chmod +x start.sh

# Use dumb-init to handle PID 1 signals properly
ENTRYPOINT ["dumb-init", "./start.sh"]
CMD ["node", "dist/src/main.js"]
